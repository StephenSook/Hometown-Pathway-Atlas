export interface RegionResponse {
  fips: string;
  county_name: string;
  state: string;
  msa_label: string;
  population: number;
  centroid: Centroid;
  metrics: {
    olympic: MetricsEntry;
    paralympic: MetricsEntry;
  };
  top_sports: SportEntry[];
  climate: ClimateBlock;
  adaptive_access: AdaptiveAccessBlock;
  narrative: string;
  narrative_source?: 'gemini' | 'fallback';
  compliance_log: ComplianceLogEntry[];
}

// "gemini" = real Vertex call. "fallback" =
// deterministic _fallback_region_narrative.
// Mirrors Pydantic Literal — no contract drift.
